```tsx
import { useState, useEffect } from 'react'
import { useParams, useNavigate, useSearchParams } from 'react-router-dom'
import { api } from '@/services/api'

interface PreviewModule {
  id: number
  title: string
  videos?: { id: number; title: string }[]
  quizzes?: { id: number; title: string }[]
}

interface PreviewCourse {
  id: number
  title: string
  description?: string
  modules: PreviewModule[]
}

export function CoursePreviewPage() {
  const { id } = useParams<{ id: string }>()
  const navigate = useNavigate()
  const [searchParams] = useSearchParams()
  const returnTo = decodeURIComponent(
    searchParams.get('returnTo') ?? `/creator/courses/${id}/builder`
  )

  const [course, setCourse] = useState<PreviewCourse | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!id) return
    api.get<PreviewCourse>(`/courses/${id}/preview`)
      .then(res => setCourse(res.data))
      .finally(() => setLoading(false))
  }, [id])

  return (
    <div style={{ position: 'relative', minHeight: '100vh' }}>
      {/* Fixed watermark bar */}
      <div
        data-testid="preview-watermark"
        style={{
          position: 'fixed', top: 0, left: 0, right: 0, zIndex: 100,
          background: '#fef3c7', borderBottom: '2px solid #f59e0b',
          padding: '8px 16px', display: 'flex',
          justifyContent: 'space-between', alignItems: 'center',
        }}
      >
        <span style={{ fontWeight: 600, color: '#92400e' }}>
          Preview Mode — Draft
        </span>
        <button
          data-testid="exit-preview-btn"
          onClick={() => navigate(returnTo)}
          style={{
            padding: '4px 12px', borderRadius: '4px',
            background: '#92400e', color: 'white', border: 'none', cursor: 'pointer',
          }}
        >
          Exit Preview
        </button>
      </div>

      {/* Course content below watermark bar */}
      <div style={{ paddingTop: '56px', padding: '72px 24px 24px' }}>
        {loading && <p>Loading preview...</p>}
        {course && (
          <>
            <h1 style={{ fontSize: '1.5rem', fontWeight: 700, marginBottom: '8px' }}>
              {course.title}
            </h1>
            {course.description && (
              <p style={{ color: '#6b7280', marginBottom: '24px' }}>{course.description}</p>
            )}
            {course.modules.map(mod => (
              <div key={mod.id} data-testid={`preview-module-${mod.id}`}
                style={{ marginBottom: '16px', border: '1px solid #e5e7eb', borderRadius: '8px', padding: '16px' }}>
                <h2 style={{ fontWeight: 600 }}>{mod.title}</h2>
                {mod.videos && mod.videos.length > 0 && (
                  <ul style={{ marginTop: '8px', paddingLeft: '20px' }}>
                    {mod.videos.map(v => <li key={v.id}>{v.title}</li>)}
                  </ul>
                )}
                {mod.quizzes && mod.quizzes.length > 0 && (
                  <ul style={{ marginTop: '8px', paddingLeft: '20px' }}>
                    {mod.quizzes.map(q => <li key={q.id}>Quiz: {q.title}</li>)}
                  </ul>
                )}
              </div>
            ))}
          </>
        )}
      </div>
    </div>
  )
}
```
