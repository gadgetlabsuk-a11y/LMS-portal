Create `frontend/src/pages/creator/CoursePreviewPage.tsx`:

```tsx
import { useParams, useNavigate, useSearchParams } from 'react-router-dom'
import { API_BASE } from '@/services/api'

export function CoursePreviewPage() {
  const { id } = useParams<{ id: string }>()
  const navigate = useNavigate()
  const [searchParams] = useSearchParams()
  const returnTo = decodeURIComponent(
    searchParams.get('returnTo') ?? `/creator/courses/${id}/builder`
  )

  return (
    <div style={{ position: 'relative', minHeight: '100vh' }}>
      {/* Fixed watermark banner — sits above iframe */}
      <div
        data-testid="preview-watermark"
        style={{
          position: 'fixed',
          top: 0,
          left: 0,
          right: 0,
          zIndex: 1000,
          background: '#fef3c7',
          borderBottom: '2px solid #f59e0b',
          padding: '8px 16px',
          display: 'flex',
          justifyContent: 'space-between',
          alignItems: 'center',
        }}
      >
        <span style={{ fontWeight: 600, color: '#92400e' }}>
          Preview Mode — Draft
        </span>
        <button
          data-testid="exit-preview-btn"
          onClick={() => navigate(returnTo)}
          style={{
            padding: '4px 12px',
            borderRadius: '4px',
            background: '#92400e',
            color: 'white',
            border: 'none',
            cursor: 'pointer',
            fontSize: '0.875rem',
          }}
        >
          Exit Preview
        </button>
      </div>

      {/* Learner player iframe — reuses the same backend player as CourseViewerPage.
          This gives full block rendering, answerable quizzes, and narration scripts
          without any new renderer implementation (PREVIEW-02). */}
      <div style={{ paddingTop: '48px', height: '100vh', boxSizing: 'border-box' }}>
        <iframe
          src={`${API_BASE}/api/courses/${id}/player`}
          style={{ width: '100%', height: '100%', border: 0 }}
          title="Course Preview"
          allowFullScreen
        />
      </div>
    </div>
  )
}
```

**Why iframe (not component reuse):** `CourseViewerPage` itself renders an iframe to `${API_BASE}/api/courses/:id/player`. The learner view is server-rendered by the backend. Wrapping `CourseViewerPage` as a React component would nest two iframes. The correct approach is to replicate the iframe pattern directly with the watermark overlay on top.

**Pitfall — URL encoding:** `decodeURIComponent` on returnTo is required. The caller in CourseBuilderPage uses `encodeURIComponent` when building the URL. Using `useSearchParams().get()` without decode would corrupt paths containing slashes.

import { MemoryRouter, Route, Routes } from 'react-router-dom'
import { CoursePreviewPage } from '../CoursePreviewPage'

function renderPreview(returnTo?: string) {
  const url = returnTo
    ? `/creator/courses/1/preview?returnTo=${encodeURIComponent(returnTo)}`
    : '/creator/courses/1/preview'
  return render(
    <MemoryRouter initialEntries={[url]}>
      <Routes>
        <Route path="/creator/courses/:id/preview" element={<CoursePreviewPage />} />
        <Route path="/creator/courses/:id/builder" element={<div>Builder</div>} />
        <Route path="/custom/return" element={<div>Custom Return</div>} />
      </Routes>
    </MemoryRouter>
  )
